# Spanish Translation Fixes - custom localization
# The Spanish localization writes [Character.GetTitledFirstName|l] so that the title is lowercase ("la condesa Bertha").
# When the character has no title, that lowercased the name itself ("bertha"), and the article was kept ("la bertha").
# These entries lowercase the title only when the character shows one, and drop the article otherwise ("Bertha", "de Bertha").
# "Has a title" follows the same rule the vanilla Spanish ES_only_el_GetTitledFirstName uses: own primary title, or a titled spouse.

STF_GetTitledFirstName = {
	type = character
	log_loc_errors = no
	text = {
		trigger = {
			OR = {
				exists = primary_title
				any_spouse = {
					exists = primary_title
					primary_title.tier >= tier_barony
				}
			}
		}
		localization_key = STF_GetTitledFirstName_titled
	}
	text = {
		localization_key = STF_GetTitledFirstName_untitled
		fallback = yes
	}
}
